Simple and Declarative

migration create_users_table {
  up {
    create_table('users') {
      id = primary_key()
      name = string(max_length: 255)
      email = string(unique: true, max_length: 255)
      created_at = timestamp()
    }
  }

  down {
    drop_table('users')
  }
}

// Run with: migratedb up
